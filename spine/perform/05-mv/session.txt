CREATE OR REPLACE MATERIALIZED VIEW MV_DAILY_REVENUE AS
SELECT order_date_key, SUM(amount_eur) AS revenue_eur, COUNT(*) AS lines
FROM FCT_ORDER_LINES WHERE order_status = 'paid'
GROUP BY order_date_key;
